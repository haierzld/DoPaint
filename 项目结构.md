# DoPaint 后端项目结构

```
backend/
│
├── main.py                          # FastAPI 应用入口
├── requirements.txt                 # Python 依赖
├── .env.example                     # 环境变量模板
├── alembic.ini                      # 数据库迁移配置
│
├── app/
│   ├── __init__.py
│   │
│   ├── core/                        # 核心配置模块
│   │   ├── __init__.py
│   │   ├── config.py                # 应用配置（读取.env）
│   │   ├── security.py              # JWT 认证 + 微信登录
│   │   └── deps.py                  # 依赖注入（数据库会话、当前用户）
│   │
│   ├── models/                      # SQLAlchemy 数据模型
│   │   ├── __init__.py
│   │   ├── base.py                  # 基础模型类
│   │   ├── organization.py          # 机构（幼儿园）
│   │   ├── user.py                  # 用户（教师/园长）
│   │   ├── artwork.py               # 画作
│   │   ├── animation.py             # 动画生成记录
│   │   ├── prompt_template.py       # 提示词模板
│   │   └── order.py                 # 订单
│   │
│   ├── schemas/                     # Pydantic 请求/响应模型
│   │   ├── __init__.py
│   │   ├── auth.py                  # 登录相关
│   │   ├── artwork.py               # 画作相关
│   │   ├── animation.py             # 动画相关
│   │   ├── organization.py          # 机构相关
│   │   ├── prompt.py                # 提示词模板相关
│   │   └── order.py                 # 订单相关
│   │
│   ├── api/                         # API 路由
│   │   ├── __init__.py
│   │   ├── router.py                # 路由汇总
│   │   ├── endpoints/
│   │   │   ├── __init__.py
│   │   │   ├── auth.py              # 登录/注册
│   │   │   ├── artworks.py          # 画作上传/管理
│   │   │   ├── animations.py        # 动画生成/查询
│   │   │   ├── prompts.py           # 提示词模板
│   │   │   ├── organizations.py     # 机构管理
│   │   │   ├── orders.py            # 订单/支付
│   │   │   └── admin.py             # 管理后台
│   │   └── deps.py                  # 路由级依赖
│   │
│   ├── services/                    # 业务逻辑层
│   │   ├── __init__.py
│   │   ├── artwork_service.py       # 画作处理服务
│   │   ├── animation_service.py     # 动画生成服务
│   │   ├── ai_service.py            # 阿里万象 API 调用
│   │   ├── quota_service.py         # 配额管理
│   │   ├── payment_service.py       # 微信支付
│   │   └── storage_service.py       # OSS 存储
│   │
│   ├── utils/                       # 工具函数
│   │   ├── __init__.py
│   │   ├── image_processor.py       # 图像预处理
│   │   ├── prompt_builder.py        # 提示词组装
│   │   ├── video_utils.py           # 视频处理
│   │   └── response.py              # 统一响应格式
│   │
│   └── tasks/                       # 异步任务
│       ├── __init__.py
│       └── animation_task.py        # 动画生成异步任务
│
├── alembic/                         # 数据库迁移
│   ├── env.py
│   └── versions/
│
├── tests/                           # 测试
│   ├── __init__.py
│   ├── test_auth.py
│   ├── test_artworks.py
│   └── test_animations.py
│
└── scripts/                         # 运维脚本
    ├── init_db.py                   # 初始化数据库
    └── seed_prompts.py              # 初始化提示词数据
```
